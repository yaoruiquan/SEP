```typescript
try {
  for await (const rawChunk of result.fullStream) {
    // ... 处理 chunk
  }
} catch (streamError) {
  this.logger.error(`Stream consumption error:`, streamError);
  yield {
    event: 'error',
    data: { message: `AI 回复生成失败: ${streamError.message}` }

  await this.prisma.message.create({
    data: { sessionId, role: 'ASSISTANT', content: '❌ 对话生成失败,请稍后重试' },

  break;
}
```
